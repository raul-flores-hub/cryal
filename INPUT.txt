# =============================================================================
# CrYAL — Crystal Structure Prediction via Active Learning
# INPUT.txt — Main configuration file
#
# Format:
#   % SECTION_NAME       starts a section
#   KEY = VALUE          parameter assignment (inline # comments allowed)
#   blank lines and lines starting with # are ignored
#
# IMPORTANT — blind search design:
#   This tool is designed to find crystal structures WITHOUT prior knowledge
#   of the experimental structure. Do NOT set betaMode, preferred SG, or
#   volume ranges based on experimental data. The GP will discover favorable
#   regions autonomously through the active learning cycles.
# =============================================================================

% GENERAL
moleculeFile          = examples/benzene.xyz   # XYZ file of one molecule (this is only the shipped example)
Z                     = 4              # molecules per unit cell (benzene crystallizes Pbca, Z = 4)
randomSeed            = 33936
outputDir             = AL_results     # all output goes here
logLevel              = INFO           # INFO or DEBUG
resume                = false          # true (or pass --resume) = continue an interrupted run in outputDir

% ENERGY_BACKEND
# Which engine relaxes each candidate and returns its energy.
#   lammps_mace  external LAMMPS process driving MACE-OFF23 (the published run)
#   ase          in-process relaxation with any ASE calculator (see below)
#   uma          in-process relaxation with UMA (fairchem); see % BACKEND_UMA
# Omitting the key means lammps_mace, so inputs written before backends
# existed keep their exact meaning.
energyBackend         = lammps_mace
energySanityMax       = -100.0         # eV per cell; energies above this = failed relaxation ('none' disables)

% BACKEND_LAMMPS
lammpsInput           = in_v3.lammps   # LAMMPS input script for relaxation
lammpsCommand         = lmp -k on g 1 -sf kk -pk kokkos newton on neigh half   # Kokkos GPU build required by MACE-OFF mliap (forward_exchange)
specOrder             = C H            # element order for LAMMPS atom types; must match pair_coeff in the LAMMPS script

% BACKEND_ASE
# Only read when energyBackend = ase. The calculator is named by dotted path,
# so no calculator is a dependency of CrYAL; it is built once and reused.
# NOTE: this is a local relaxation, not the MD-NPT protocol of the LAMMPS
# script, and energies from different backends are not comparable.
# aseCalculator       = mace.calculators.mace_off
# aseCalculatorKwargs = model=medium device=cuda
# aseOptimizer        = FIRE           # FIRE, BFGS, LBFGS, BFGSLineSearch, GPMin
# aseFmax             = 0.05           # eV/A
# aseMaxSteps         = 500            # a relaxation that does not converge counts as failed
# aseRelaxCell        = true           # relax cell and positions (FrechetCellFilter)

% BACKEND_UMA
# Only read when energyBackend = uma. UMA needs no dotted path and no factory
# module of your own: it ships as a backend because it cannot be named by a
# dotted path (its one-call constructor is a classmethod) and because a
# hand-written factory would not exist on the other machines of a parallel run.
#
# Install: `pip install fairchem-core` into the interpreter that runs CrYAL,
# plus a Hugging Face token for the gated checkpoint. This is NOT the ORCA
# external-tools installation — CrYAL drives UMA in-process through ASE and
# needs neither the Flask server nor the ORCA client. See INSTALL.md §5.
#
# umaTask is MANDATORY and has no default. Each head was trained on a different
# domain and each has its own zero of energy: for one periodic C50Au4 cell
# 'omat' gives -466.55 eV and 'oc20' -447.06 eV. Numbers from two heads are not
# comparable and nothing downstream would show the mix.
#   omc    organic molecular crystals   <- this is the one for CSP
#   omat   inorganic / periodic materials
#   oc20   adsorbates on catalyst surfaces
#   omol   isolated molecules (not periodic)
#   odac   MOFs for direct air capture
# umaModel      = uma-s-1p1     # checkpoint; older fairchem does not list 1p2
# umaTask       = omc           # MANDATORY, no default
# umaDevice     = cuda          # cuda or cpu
# umaInferenceSettings = default   # 'default' or a preset name; do NOT build
#                                  # an InferenceSettings object by hand
# The relaxation itself is configured by the aseOptimizer / aseFmax /
# aseMaxSteps / aseRelaxCell keys above, shared with the ase backend.

% ACTIVE_LEARNING
numCycles             = 10             # number of AL cycles after initialization
initialStructures     = 100            # structures generated in cycle 0
structuresPerCycle    = 50             # structures generated per AL cycle
poolSize              = 20             # best structures kept as seeds between cycles
maxNoImproveCycles    = 3              # stop if best energy unchanged for N cycles

% SPACE_GROUPS
# Weights based on CSD frequency statistics for organic molecular crystals.
# No bias toward any specific SG — the GP will discover which is favorable.
#   SG 14 (P2_1/c):   ~36%   monoclinic
#   SG  2 (P-1):      ~24%   triclinic
#   SG 19 (P2_12_12_1): ~11% orthorhombic
#   SG 15 (C2/c):      ~7%   monoclinic
#   SG  4 (P2_1):      ~6%   monoclinic
#   SG  9 (Cc):        ~4%   monoclinic
#   SG 33 (Pna2_1):    ~2%   orthorhombic
#   SG 61 (Pbca):      ~2%   orthorhombic
spaceGroups           = 14 2 19 15 4 9 33 61
sgWeights             = 36 24 11  7 6 4  2  2

% VOLUME
# Volume range in Å³ for initial structure generation.
# Use 'auto' to estimate from molecular weight + density range below.
# This is the only prior knowledge needed: a plausible density range
# for organic molecular crystals (typically 0.8–1.6 g/cm³).
targetVolumeMin       = auto
targetVolumeMax       = auto
densityMin            = 0.8            # g/cm³ — lower bound for auto estimate
densityMax            = 1.6            # g/cm³ — upper bound for auto estimate
# Expansion factors: if PyXtal cannot place molecules at target volume,
# the cell is expanded in these steps until placement succeeds (USPEX strategy).
volumeExpansionFactors = 1.0 1.05 1.10 1.15 1.20 1.35 1.50

% CELL
minCellAxis           = 5.0            # Å — reject cells with any axis < this
                                       # Scale this with your molecule: 5 Å suits benzene
                                       # (real cell 7.4 x 9.4 x 6.8 Å); a large planar
                                       # molecule needs a much larger value.

% BETA_MONOCLINIC
# Beta angle distribution for monoclinic space groups — UNIFORM (blind).
# The GP + EI will discover the favorable beta range through sampling.
# Do NOT set distribution = triangular unless you have prior knowledge.
betaMin               = 90.0
betaMax               = 150.0
betaDistribution      = uniform        # 'uniform' (blind) or 'triangular' (informed)
# betaMode            = 101.0         # only used if betaDistribution = triangular

% GP
gpKernel              = matern25       # 'matern25' (recommended) or 'rbf'
gpNRestarts           = 5              # optimizer restarts for kernel hyperparameters
eiXi                  = 0.01          # exploration parameter (higher = more exploration)
numEICandidates       = 50000         # random candidates evaluated for EI maximization
numEISuggestions      = 20            # top EI candidates used to bias next generation

% CHPI_OPTIMIZER
# CH-pi geometric optimizer: adjusts molecular positions to maximize
# H...aromatic centroid contacts in [dMin, dMax] range.
# Set useChpiOptimizer = False to skip this step entirely.
useChpiOptimizer      = True
dMin                  = 2.6           # Å — equilibrium range lower bound
dMax                  = 3.5           # Å — equilibrium range upper bound
dTarget               = 3.0           # Å — target H...centroid distance
dSearch               = 6.0           # Å — search radius for contacts
angleCHCentroidMin    = 110.0         # degrees — min C-H...centroid angle
angleFromNormalMax    = 55.0          # degrees — max angle from ring normal
maxChpiIter           = 100          # max optimizer iterations per structure
maxNoImprove          = 10            # stop if no improvement for N iterations

% CONTACTS
# Minimum intermolecular contact distances for structure filtering
minHHDistance         = 1.8           # Å — H...H
minHeavyDistance      = 2.5           # Å — heavy...heavy
minHXDistance         = 1.8           # Å — H...heavy

% SEEDS
# If useSeedsFolder = True, CIF files in seedsFolder/ are loaded
# and used to generate perturbations as additional starting structures.
# If the folder is empty or does not exist, the run starts from PyXtal only.
useSeedsFolder        = True
seedsFolder           = Seeds
numPerturbationsPerSeed = 30
maxRotPerturb         = 30.0          # degrees
maxTransPerturb       = 0.1           # Å
fracMolsPerturb       = 0.5           # fraction of molecules perturbed per structure

% INTEGRITY
# Molecular integrity check after LAMMPS relaxation.
# Rejects structures where MACE-OFF23 (reactive potential) broke covalent bonds.
checkMolecularIntegrity = True
integrityMaxBondRatio   = 2.0         # bond broken if length > ratio * sum(cov_radii)

% PARALLEL
# Candidates within a cycle are independent, so they can be relaxed on several
# machines at once. This distributes whole candidates; it does not split one
# relaxation, and it does not change what any of them returns.
#
# Requirements on every worker: passwordless ssh from this machine, CrYAL and
# its dependencies installed, and the energy engine (lmp) available to a
# non-interactive ssh session.
#
# parallelWorkers  one entry per machine — user@host:slots
#                  slots = how many candidates that machine relaxes at once.
#                  Add a third field when the interpreter that can import
#                  cryal is not the `python3` on its PATH, which on a machine
#                  set up for MACE it usually is not:
#                      raul@192.168.1.11:4:/home/raul/mace-env/bin/python
#                  Machines pull candidates from a shared queue, so a slow one
#                  simply takes fewer; nothing waits on the slowest.
useParallel           = false
parallelWorkers       =
parallelLocalSlots    = 1             # slots on the server itself
parallelRemoteDir     = ~/.cryal_work # per-run work dir on each worker
parallelPython        = python3       # default interpreter for the workers
parallelSshKey        =               # private key, if not the default one
parallelSshTimeout    = 30            # s — give up on connecting
parallelTaskTimeout   = 0             # s — 0 derives it from the backend
parallelMaxFailures   = 3             # drop a machine after N failures in a row
parallelKeepRemote    = false         # leave the remote work dir for debugging
parallelFetchOnFailure = true         # copy a failed candidate's files back
parallelFetchStep     = false         # copy every candidate's files back
